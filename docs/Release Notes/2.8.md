## Appendix

* Python code to fix naming

```python
import dataiku

old_name = "Mazzei_Designer"
new_name = "mazzei_designer"

folder = dataiku.Folder("partitioned_data", project_key=dataiku.default_project_key())

paths = folder.list_paths_in_partition()
filtered_paths = [
    p for p in paths
    if old_name in p
]

failed = []
for old_path in filtered_paths:
    new_path = old_path.replace(old_name, new_name)
    try:
        with folder.get_download_stream(old_path) as stream:
            folder.upload_stream(new_path, stream)
    except:
        print(f"failed to update path for {old_path}")
        failed.append(old_path)
        continue
    folder.delete_path(old_path)
```
