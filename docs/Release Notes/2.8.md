for old_path in filtered_paths:
    new_path = old_path.replace(old_name, new_name)
    try:
        with folder.get_download_stream(old_path) as stream:
            folder.upload_stream(new_path, stream)
    except:
        print(f"failed to update path for {old_path}")
        failed.append(old_path)
        continue
